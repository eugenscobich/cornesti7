script:
  - id: dimmer_hold
    mode: restart
    parameters:
      input: esphome::binary_sensor::BinarySensor*
      light: esphome::light::LightState*
    then:
      - while:
          condition:
            lambda: |
              return input->state;
          then:
            - lambda: |
                // Current brightness in [0.0, 1.0]
                float b = light->current_values.get_brightness();
                //ESP_LOGD("app", "b: %f", b);

                // Step size per tick
                const float step = 0.02f;  // ~2% every cycle

                // Apply direction
                if (id(light_dimmer_dir) > 0) {
                  b += step;
                } else {
                  b -= step;
                }
                // Clamp + reverse at endpoints
                if (b >= 1.0f) { b = 1.0f; id(light_dimmer_dir) = -1; }
                if (b <= 0.05f){ b = 0.05f; id(light_dimmer_dir) =  1; } // keep a faint minimum

                // Send the change with a short transition for smoothness
                auto call = light->turn_on();
                call.set_brightness(b);
                call.set_transition_length(150); // ms
                call.perform();
            - delay: 150ms
